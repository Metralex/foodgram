# Третий проект - eda.hopto.org
server {
    listen 80;
    server_name eda.hopto.org;
    # Перенаправление на HTTPS
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name eda.hopto.org;
    ssl_certificate /etc/letsencrypt/live/eda.hopto.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/eda.hopto.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
    
    client_max_body_size 30M;

    location / {
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;  # важно для CSRF через HTTPS
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Server $host;
        proxy_pass http://127.0.0.1:10000;           # внутренний порт foodgram nginx контейнера
    }
}
